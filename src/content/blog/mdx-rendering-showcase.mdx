Body text uses LXGW WenKai Screen, with English text, numbers and common symbols rendered in harmony by the same font; code blocks keep their own independent monospace font stack.

export const InfoBadge = ({ text }) => (
	<span
		style={{
			display: 'inline-block',
			padding: '0.15rem 0.55rem',
			borderRadius: 'var(--radius-ui)',
			fontSize: 'var(--text-xs)',
			background: 'rgba(99, 102, 241, 0.14)',
			color: 'rgb(79, 70, 229)',
		}}
	>
		{text}
	</span>
);
